322: function useRedirectMessage(setMsg, okText) {
323:   useEffect(() => {
324:     let active = true;
325:     resolveRedirectOutcome().then(({ ok, result, error, cameBack }) => {
326:       if (!active) return;
327:       if (!ok) {
328:         console.error('[Auth] link redirect error:', error?.code, error?.message);
329:         if (!IGNORED.includes(error?.code)) setMsg({ type: 'err', text: authErrorMessage(error?.code) });
330:         return;
331:       }
332:       console.log('[Auth] link redirect result:', result);
333:       if (result && result.operationType === 'link') setMsg({ type: 'ok', text: okText });
334:       else if (!result && cameBack) setMsg({ type: 'err', text: 'Die Verknüpfung konnte nicht abgeschlossen werden. Bitte erneut versuchen.' });
335:     });
336:     return () => { active = false; };
337:     // eslint-disable-next-line react-hooks/exhaustive-deps
338:   }, []);
