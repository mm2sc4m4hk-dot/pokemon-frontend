import React from 'react';
import { navigate } from './router';

export const BRAND_NAME = 'TCG Vault';
export const DISCLAIMER = 'Inoffizielles Fan-Projekt. Nicht mit The Pokémon Company, Nintendo oder Creatures Inc. verbunden.';

export function Disclaimer({ className = '' }) {
  return (
    <p className={`text-[10px] leading-snug text-slate-500 text-center ${className}`}>
      {DISCLAIMER}
    </p>
  );
}

// Links zu den Rechtsseiten (Impressum/Datenschutz). Werden per pushState
// aufgerufen, damit die App nicht neu geladen werden muss.
export function LegalLinks({ className = '' }) {
  const link = 'text-[11px] text-slate-400 hover:text-cyan-400 underline underline-offset-2 transition-colors';
  return (
    <nav className={`flex items-center justify-center gap-3 flex-wrap ${className}`}>
      <button type="button" className={link} onClick={() => navigate('/impressum')}>Impressum</button>
      <span className="text-slate-700 text-[11px]">·</span>
      <button type="button" className={link} onClick={() => navigate('/datenschutz')}>Datenschutz</button>
    </nav>
  );
}
