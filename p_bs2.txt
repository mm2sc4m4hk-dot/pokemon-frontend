28: // Raster über das Foto legen (trim = Rand in %, der auf allen Seiten abgeschnitten wird)
29: function cellRects(w, h, rows, cols, trim) {
30:   const tx = (w * trim) / 100;
31:   const ty = (h * trim) / 100;
32:   const cw = (w - 2 * tx) / cols;
33:   const ch = (h - 2 * ty) / rows;
34:   const out = [];
35:   for (let r = 0; r < rows; r += 1) {
36:     for (let c = 0; c < cols; c += 1) out.push({ x: tx + c * cw, y: ty + r * ch, w: cw, h: ch });
37:   }
38:   return out;
39: }

42: function cellCanvas(source, rect) {
43:   const { sx, sy, sw } = coverRect(rect.w, rect.h);
44:   const sh = (sw * 7) / 5;
45:   const outW = Math.min(800, Math.round(sw));
46:   const c = document.createElement('canvas');
47:   c.width = outW;
48:   c.height = Math.round((outW * 7) / 5);
49:   c.getContext('2d').drawImage(source, rect.x + sx, rect.y + sy, sw, Math.min(sh, rect.h), 0, 0, c.width, c.height);
50:   return c;
51: }
