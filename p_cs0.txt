22: export const loadImageSource = (file) =>
23:   new Promise((resolve, reject) => {
24:     const reader = new FileReader();
25:     reader.onerror = reject;
26:     reader.onload = (e) => {
27:       const img = new Image();
28:       img.onerror = reject;
29:       img.onload = () => resolve({ source: img, w: img.naturalWidth || img.width, h: img.naturalHeight || img.height });
30:       img.src = e.target.result;
31:     };
32:     reader.readAsDataURL(file);
33:   });

35: // Ein Versuch. Der Server wiederholt bei Überlastung selbst.
36: async function readCardOnce(base64Image, api) {
37:   const controller = new AbortController();
38:   const timer = setTimeout(() => controller.abort(), 90000); // Render-Gratisplan kann schlafen
39:   try {
40:     const user = auth.currentUser;
41:     if (!user) { const e = new Error('Bitte melde dich an, um zu scannen.'); e.final = true; throw e; }
42:     let token;
43:     try { token = await user.getIdToken(); } catch (err) {
44:       const e = new Error('Die Anmeldung konnte nicht erneuert werden. Internet prüfen oder neu anmelden.'); e.busy = true; throw e;
45:     }
46:     const response = await fetch(`${api}/api/scan-genai`, {
47:       method: 'POST',
48:       headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` },
49:       signal: controller.signal,
50:       body: JSON.stringify({ image: base64Image })
51:     });
52:     const data = await response.json().catch(() => ({}));
53:     if (response.status === 401) { const e = new Error('Anmeldung abgelaufen oder ungültig. Bitte neu anmelden.'); e.final = true; throw e; }
54:     if (response.status === 429) { const e = new Error('Zu viele Scans auf einmal. Bitte kurz warten.'); e.final = true; throw e; }
55:     if (response.status === 503) { const e = new Error('Der Scan ist gerade ausgelastet. Bitte warte kurz und scanne erneut.'); e.busy = true; throw e; }
56:     if (!response.ok) { const e = new Error(data.error || `Fehler beim Scannen (Status ${response.status}).`); e.final = true; throw e; }
57:     return data;
58:   } catch (e) {
59:     if (e.name === 'AbortError') { const t = new Error('Der Server hat nicht geantwortet (schläft evtl.). Bitte gleich nochmal versuchen.'); t.busy = true; throw t; }
60:     if (e instanceof TypeError) { const t = new Error('Keine Verbindung zum Server. Internet prüfen und erneut versuchen.'); t.busy = true; throw t; }
61:     throw e;
62:   } finally {
63:     clearTimeout(timer);
64:   }
65: }

67: // Mit einer automatischen Wiederholung (nach 3 s) bei Überlastung/Netzfehler
68: export const readCard = async (base64Image, api = API_URL, { retries = 1 } = {}) => {
69:   for (let attempt = 0; ; attempt += 1) {
70:     try {
71:       return await readCardOnce(base64Image, api);
72:     } catch (e) {
73:       if (!e.busy || attempt >= retries) throw e;
74:       await sleep(3000);
75:     }
76:   }
77: };

79: // Foto verkleinern (schneller Upload, unter dem 10-MB-Limit des Servers)
80: function shrinkToCanvas(source, w, h, maxSide = 900) {
81:   const scale = Math.min(1, maxSide / Math.max(w, h));
82:   const c = document.createElement('canvas');
83:   c.width = Math.round(w * scale);
84:   c.height = Math.round(h * scale);
85:   c.getContext('2d').drawImage(source, 0, 0, c.width, c.height);
86:   return c;
